cmake_minimum_required(VERSION 3.20)
project(calendar_recovered LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(calendar_recovered STATIC calendar.cpp)
add_executable(calendar_tests calendar_tests.cpp)
target_link_libraries(calendar_tests PRIVATE calendar_recovered)
enable_testing()
add_test(NAME calendar_checks COMMAND calendar_tests)
