cmake_minimum_required(VERSION 3.20)
project(anadius_recovered_bootstrap LANGUAGES CXX)
add_library(anadius_bootstrap STATIC bootstrap.cpp)
target_compile_features(anadius_bootstrap PUBLIC cxx_std_20)
target_include_directories(anadius_bootstrap PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bootstrap_contract_tests bootstrap_tests.cpp)
target_link_libraries(bootstrap_contract_tests PRIVATE anadius_bootstrap)
enable_testing()
add_test(NAME bootstrap_contracts COMMAND bootstrap_contract_tests)
