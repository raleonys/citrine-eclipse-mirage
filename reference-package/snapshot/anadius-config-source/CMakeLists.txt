cmake_minimum_required(VERSION 3.20)
project(anadius_recovered_configuration LANGUAGES CXX)
add_library(anadius_configuration STATIC configuration.cpp)
target_compile_features(anadius_configuration PUBLIC cxx_std_20)
target_include_directories(anadius_configuration PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(configuration_contract_tests configuration_tests.cpp)
target_link_libraries(configuration_contract_tests PRIVATE anadius_configuration)
enable_testing()
add_test(NAME configuration_contracts COMMAND configuration_contract_tests)
