cmake_minimum_required(VERSION 3.20)
project(anadius_server_reconstruction LANGUAGES CXX)
add_library(anadius_server STATIC server.cpp ../anadius-execution-source/execution.cpp)
target_compile_features(anadius_server PUBLIC cxx_std_20)
target_include_directories(anadius_server PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(server_mock_tests server_tests.cpp)
target_link_libraries(server_mock_tests PRIVATE anadius_server)
target_compile_features(server_mock_tests PRIVATE cxx_std_20)
enable_testing()
add_test(NAME server_mock_tests COMMAND server_mock_tests)
