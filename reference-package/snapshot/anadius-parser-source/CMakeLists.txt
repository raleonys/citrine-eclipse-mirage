cmake_minimum_required(VERSION 3.20)
project(anadius_parser_reconstruction LANGUAGES CXX)
add_library(anadius_parser STATIC parser.cpp ../anadius-config-source/configuration.cpp)
target_compile_features(anadius_parser PUBLIC cxx_std_20)
target_include_directories(anadius_parser PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(parser_tests parser_tests.cpp)
target_link_libraries(parser_tests PRIVATE anadius_parser)
enable_testing()
add_test(NAME parser_synthetic_contracts COMMAND parser_tests)
