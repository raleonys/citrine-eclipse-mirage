file(READ "${CMAKE_CURRENT_SOURCE_DIR}/metadata/BUILD_GRAPH.json" graph)
function(read_array key output)
  string(JSON count LENGTH "${graph}" "${key}")
  math(EXPR last "${count} - 1")
  set(items "")
  foreach(i RANGE 0 ${last})
    string(JSON item GET "${graph}" "${key}" ${i})
    list(APPEND items "${item}")
  endforeach()
  set(${output} "${items}" PARENT_SCOPE)
endfunction()

    string(JSON path GET "${graph}" tests "${name}")

    if(MSVC)
      target_compile_options(${name} PRIVATE /UNDEBUG)
    else()

    set(test_arguments "")
    string(JSON argument_count ERROR_VARIABLE argument_error LENGTH "${graph}" test_arguments "${name}")

      foreach(argument_index RANGE 0 ${argument_last})
        string(JSON argument GET "${graph}" test_arguments "${name}" ${argument_index})
        list(APPEND test_arguments "${CMAKE_CURRENT_SOURCE_DIR}/${argument}")
      endforeach()

  endforeach()
