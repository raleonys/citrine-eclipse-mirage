namespace {
int failure=0,calls=0,allocated=0,released=0;
void reset(int value) { failure=value;calls=allocated=released=0; }
}
namespace preloader_l::platform::host_policy_detail {
void* heap_alloc(size_t n) noexcept {
    if(failure==2) return nullptr;
    void* p=std::malloc(n);if(p)++allocated;return p;
}
void heap_free(void* p) noexcept { if(p)++released;std::free(p); }
int32_t write_w(preloader_l::config::ValueView,const uint16_t*,size_t,uint32_t,uint32_t*,uint8_t*,uint32_t*) noexcept;
}
extern "C" int MultiByteToWideChar(unsigned int cp,unsigned long flags,
                                  const char* input,int length,uint16_t* output,int capacity) {
    ++calls;assert(cp==65001 && flags==0 && input && length==1 && input[0]=='A');
    if(failure==1 || (failure==3 && output)) return 0;
    if(output) {assert(capacity==1);output[0]=0x20ac;}
    return 1;
}
int main() {
    using namespace preloader_l;
    using namespace preloader_l::platform;
    using namespace preloader_l::platform::host_policy_detail;
    const uint16_t fallback[]={0x42};
    const config::ValueView configured{"A",1,true};
    for(int mode=0;mode<=3;++mode) {
        reset(mode);alignas(8) uint8_t output[8]={0xa5,0xa5,0xa5,0xa5,0xa5,0xa5,0xa5,0xa5};
        uint32_t type=99,bytes=99;
        const auto result=write_w(configured,fallback,1,4,&type,output+1,&bytes);
        if(mode==0) {
            assert(result==host_error_success && calls==2 && allocated==1 && released==1);
            assert(type==host_reg_sz && bytes==4 && output[1]==0xac && output[2]==0x20 && output[3]==0 && output[4]==0);
        } else {
            assert(result==host_error_invalid_parameter && type==99 && bytes==99);
            assert(calls==(mode==3?2:1));assert(allocated==(mode==3?1:0) && released==allocated);
            for(auto byte:output) assert(byte==0xa5);
        }
        assert(output[0]==0xa5 && output[5]==0xa5);
    }
    reset(0);uint8_t output[8]={};uint32_t bytes=99,type=99;
    assert(write_w({nullptr,0,true},fallback,1,2,&type,output+1,&bytes)==host_error_success);
    assert(calls==0 && allocated==0 && released==0 && bytes==2 && type==host_reg_sz);
    reset(0);type=99;bytes=99;
    assert(write_w(configured,fallback,1,4,&type,output+1,nullptr)==host_error_invalid_parameter);
    assert(calls==2 && allocated==1 && released==1 && type==99 && bytes==99);
    reset(0);type=99;bytes=99;
    assert(write_w(configured,fallback,1,3,&type,output+1,&bytes)==host_error_more_data);
    assert(calls==2 && allocated==1 && released==1 && bytes==4 && type==host_reg_sz);
    reset(0);type=99;bytes=99;
    assert(write_w(configured,fallback,1,0,&type,nullptr,&bytes)==host_error_success);
    assert(calls==2 && allocated==1 && released==1 && bytes==4 && type==host_reg_sz);
}
