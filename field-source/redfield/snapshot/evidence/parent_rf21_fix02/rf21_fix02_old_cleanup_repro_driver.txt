bool cleanup_session(void*) noexcept {
    if (!hooks.stop(0)) return false;
    auto* session=pending_session.load(std::memory_order_acquire);
    if (!session) return true;
    try {
        if (!session->stop_for(std::chrono::milliseconds(0))) return false;
        pending_session.store(nullptr,std::memory_order_release);
        delete session;
        return true;

}
