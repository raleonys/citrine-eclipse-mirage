def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
lines=log.read_text(errors='replace').splitlines()
compile_runs=[s for s in lines if re.match(r'^\[RUN\] CXX \d+/170:',s)]
compile_exits=[s for s in lines if re.match(r'^\[EXIT 0; [^\]]+\] CXX \d+/170:',s)]
tests=[s for s in lines if re.match(r'^\[EXIT 0; [^\]]+\] test_.*\.exe$',s)]
assert len(compile_runs)==len(compile_exits)==170 and len(tests)==90
assert any(s.startswith('[EXIT 14;') and s.endswith('test_get_config_legacy.exe') for s in lines)
