cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(preloader_l LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PRELOADER_L_PROFILE "PRODUCTION" CACHE STRING "PRODUCTION DEBUG NATIVE CRT DIAGNOSTIC")
set_property(CACHE PRELOADER_L_PROFILE PROPERTY STRINGS PRODUCTION DEBUG NATIVE CRT DIAGNOSTIC)
if(NOT PRELOADER_L_PROFILE MATCHES "^(PRODUCTION|DEBUG|NATIVE|CRT|DIAGNOSTIC)$")
  message(FATAL_ERROR "Unknown preloader_l profile")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "preloader_l requires x64")
endif()
if(WIN32 AND NOT MSVC)
  message(FATAL_ERROR "The Windows DLL target requires MSVC x64")
endif()
if(PRELOADER_L_PROFILE MATCHES "^(DEBUG|CRT)$")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDebug")
  set(CMAKE_CONFIGURATION_TYPES Debug CACHE STRING "Profile configuration" FORCE)
else()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Profile configuration" FORCE)
endif()
find_package(Threads REQUIRED)
include(cmake/sources.cmake)
add_library(preloader_l_core STATIC ${PRELOADER_L_CORE_SOURCES})
target_include_directories(preloader_l_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(preloader_l_core SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/asio/include")
target_include_directories(preloader_l_core SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/pegtl/include")
target_include_directories(preloader_l_core SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/filesystem/include")
target_compile_definitions(preloader_l_core PRIVATE ASIO_STANDALONE=1 ASIO_NO_DEPRECATED=1)
target_link_libraries(preloader_l_core PUBLIC Threads::Threads)
if(PRELOADER_L_PROFILE MATCHES "^(DEBUG|DIAGNOSTIC)$")
  target_compile_definitions(preloader_l_core PUBLIC PRELOADER_L_DISK_DIAGNOSTICS=1)
else()
  target_compile_definitions(preloader_l_core PUBLIC PRELOADER_L_DISK_DIAGNOSTICS=0)
endif()
if(MSVC)
  target_compile_definitions(preloader_l_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
  target_compile_options(preloader_l_core PRIVATE /W4 /we4702 /EHsc /permissive- /utf-8 /GS /guard:cf)
  if(PRELOADER_L_PROFILE MATCHES "^(DEBUG|CRT)$")
    target_compile_options(preloader_l_core PRIVATE /Od /Zi /RTC1)
  else()
    target_compile_options(preloader_l_core PRIVATE /O2)
  endif()
  target_sources(preloader_l_core PRIVATE src/windows/startup_platform.cpp src/windows/library_calls.cpp src/windows/encoding_calls.cpp src/windows/status_calls.cpp)
  target_link_libraries(preloader_l_core PUBLIC ws2_32 shell32 user32 advapi32)
endif()
if(WIN32)
  add_library(preloader_l_hook_engine STATIC
    third_party/minhook/src/hook.c
    third_party/minhook/src/buffer.c
    third_party/minhook/src/trampoline.c
    third_party/minhook/src/hde/hde64.c)
  target_compile_definitions(preloader_l_hook_engine PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(preloader_l_hook_engine PRIVATE /W3 /GS /guard:cf)
  enable_language(RC)
  add_library(preloader_l SHARED ${PRELOADER_L_WINDOWS_SOURCES})
  if(PRELOADER_L_PROFILE STREQUAL "PRODUCTION")
    target_sources(preloader_l PRIVATE resources/golden_version.rc exports/production.def)
    target_compile_options(preloader_l PRIVATE /GL-)
    target_compile_options(preloader_l_core PRIVATE /GL-)
    target_compile_options(preloader_l_hook_engine PRIVATE /GL-)
    set_property(TARGET preloader_l preloader_l_core preloader_l_hook_engine PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
    target_link_options(preloader_l PRIVATE /DEBUG:NONE /LTCG:OFF /NOCOFFGRPINFO /MANIFEST:NO /SUBSYSTEM:WINDOWS,6.00)
  else()
    target_sources(preloader_l PRIVATE resources/version.rc exports/preloader_l.def)
  endif()
  target_link_libraries(preloader_l PRIVATE preloader_l_core preloader_l_hook_engine crypt32)
  target_include_directories(preloader_l PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" include)
  target_compile_options(preloader_l PRIVATE /W4 /we4702 /EHsc /permissive- /utf-8 /GS /guard:cf)
  target_link_options(preloader_l PRIVATE /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /guard:cf /INCREMENTAL:NO)
  if(PRELOADER_L_PROFILE STREQUAL "DEBUG")
    set_target_properties(preloader_l PROPERTIES OUTPUT_NAME preloader_l_debug)
  elseif(PRELOADER_L_PROFILE STREQUAL "NATIVE")
    set_target_properties(preloader_l PROPERTIES OUTPUT_NAME preloader_l_native_test)
  elseif(PRELOADER_L_PROFILE STREQUAL "CRT")
    set_target_properties(preloader_l PROPERTIES OUTPUT_NAME preloader_l_crt_test)
  elseif(PRELOADER_L_PROFILE STREQUAL "DIAGNOSTIC")
    set_target_properties(preloader_l PROPERTIES OUTPUT_NAME preloader_l_diagnostic)
  else()
    set_target_properties(preloader_l PROPERTIES OUTPUT_NAME preloader_l)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(source IN LISTS PRELOADER_L_UNIT_TEST_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    get_filename_component(parent "${source}" DIRECTORY)
    get_filename_component(group "${parent}" NAME)
    set(target "${group}_${name}")
    add_executable(${target} "${source}")
    target_link_libraries(${target} PRIVATE preloader_l_core)
    if(name STREQUAL "test_hash_table")
      target_sources(${target} PRIVATE tests/support/configuration/order.cpp)
    endif()
    if(name MATCHES "^test_registry_(source|wait)$")
      target_sources(${target} PRIVATE src/hooks/registry.cpp)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 60)
  endforeach()
  add_executable(test_runtime tests/integration/test_runtime.cpp)
  target_link_libraries(test_runtime PRIVATE preloader_l_core)
  add_test(NAME runtime_tcp_and_persistence COMMAND test_runtime)
  set_tests_properties(runtime_tcp_and_persistence PROPERTIES TIMEOUT 60)
  add_executable(test_startup_compat tests/integration/test_startup_compat.cpp)
  target_link_libraries(test_startup_compat PRIVATE preloader_l_core)
  add_test(NAME startup_configuration_profile_compat COMMAND test_startup_compat)
  set_tests_properties(startup_configuration_profile_compat PROPERTIES TIMEOUT 60)
  foreach(source IN ITEMS tests/integration/test_unload_scan.cpp tests/integration/test_atomic_stage_collision.cpp tests/integration/test_xml_depth_source_oracle.cpp tests/integration/test_config2_source_oracle.cpp tests/integration/test_failure_diagnostics.cpp tests/integration/test_ticket_output.cpp tests/integration/test_avatar_effects.cpp tests/integration/test_transport_pipeline.cpp tests/integration/test_path_effects.cpp tests/integration/test_path_resolution.cpp tests/integration/test_configuration_orchestration.cpp tests/integration/test_startup_phases.cpp tests/integration/test_brownfield_contracts.cpp tests/integration/test_supplied_config.cpp tests/integration/test_configuration_profile.cpp tests/integration/test_asio_transport.cpp tests/integration/test_filesystem_config.cpp tests/integration/test_runtime_migration.cpp tests/integration/test_original_compat.cpp)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    if(name STREQUAL "test_config2_source_oracle")
      target_sources(${name} PRIVATE tests/support/configuration/packet_parser.cpp)
    endif()
    if(name STREQUAL "test_xml_depth_source_oracle")
      target_sources(${name} PRIVATE tests/support/xml/packet_oracle.cpp)
    endif()
    if(name STREQUAL "test_atomic_stage_collision")
      target_sources(${name} PRIVATE tests/support/files/packet_binary_io.cpp)
    endif()
    if(name STREQUAL "test_unload_scan" AND NOT WIN32)
      target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests/support/unload/winshim")
      # Test-only linker observation of production memcpy; no shipping flags change.
      target_compile_options(${name} PRIVATE -fno-builtin-memcpy -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=0)
      target_link_options(${name} PRIVATE "-Wl,--wrap=memcpy")
    endif()
    target_link_libraries(${name} PRIVATE preloader_l_core)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME golden_protocol_tool_selftests COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/golden/scripts/test_pe_metadata_tools.py")
  foreach(tool_test IN ITEMS tests/tools/test_golden_pipeline.py tests/tools/test_golden_link_policy.py)
    get_filename_component(name "${tool_test}" NAME_WE)
    add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/${tool_test}")
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endforeach()
  if(MSVC AND PRELOADER_L_PROFILE STREQUAL "CRT")
    add_executable(test_crt tests/native/test_crt.cpp)
    target_link_libraries(test_crt PRIVATE preloader_l_core)
    add_test(NAME crt_lifetime_balance COMMAND test_crt)
  endif()
  if(MSVC AND PRELOADER_L_PROFILE MATCHES "^(NATIVE|CRT)$")
    add_library(online_fixture SHARED tests/native/online_fixture.cpp)
    target_include_directories(online_fixture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(online_fixture PROPERTIES OUTPUT_NAME preloader_l_online)
  endif()
endif()
