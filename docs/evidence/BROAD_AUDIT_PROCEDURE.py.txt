from pathlib import Path
import csv,json,re,hashlib,difflib
root=Path('/workspace/yellowfield');refs=Path('/workspace/re9-review/extracted');out=Path('/workspace/broad-source-audit/comparison')
rows=[{'yellowfield':r['current'],'original':r['original']} for r in json.loads(Path('/workspace/re9-review/current_re9_comparison.json').read_text())['all_pairs']]
pairs=[]
for row in rows:
 name=row['yellowfield'];original=row['original']
 if name.startswith(('src/','include/')) and (root/name).is_file() and (refs/original).is_file():pairs.append((name,original))
namespace_map={}
for current,original in pairs:
 a=re.findall(r'\bnamespace\s+([a-zA-Z_]\w*(?:::\w+)*)\s*\{',(root/current).read_text())
 b=re.findall(r'\bnamespace\s+([a-zA-Z_]\w*(?:::\w+)*)\s*\{',(refs/original).read_text())
 if a and b and a[0]!=b[0]:namespace_map[b[0]]=a[0]
lex=re.compile(r'//[^\n]*|/\*.*?\*/|(?:u8|u|U|L)?"(?:\\.|[^"\\])*"|(?:u8|u|U|L)?\'(?:\\.|[^\'\\])*\'|[a-zA-Z_]\w*|\d+(?:\.\d+)?|\S',re.S)
def normalize(s):
 s=re.sub(r'^\s*#\s*include[^\n]*','',s,flags=re.M)
 for a,b in sorted(namespace_map.items(),key=lambda x:-len(x[0])):s=s.replace(a,b)
 return [m.group(0) for m in lex.finditer(s) if not m.group(0).startswith(('//','/*'))]
results=[]
for current,original in pairs:
 a=(root/current).read_text();b=(refs/original).read_text();aa=normalize(a);bb=normalize(b)
 matched=aa==bb
 record={'current':current,'original':original,'current_sha256':hashlib.sha256((root/current).read_bytes()).hexdigest(),'original_sha256':hashlib.sha256((refs/original).read_bytes()).hexdigest(),'normalized_token_match':matched,'current_tokens':len(aa),'original_tokens':len(bb)}
 if not matched:
  name=current.replace('/','__')+'.diff';(out/name).write_text(''.join(difflib.unified_diff(b.splitlines(True),a.splitlines(True),fromfile=original,tofile=current)))
  record['diff']=name
 results.append(record)
all_current={p.relative_to(root).as_posix() for folder in ('src','include') for p in (root/folder).rglob('*') if p.is_file()}
report={'comparison':'includes/comments/whitespace removed; observed namespace renames only','namespace_map':namespace_map,'mapped_files':len(results),'normalized_matches':sum(r['normalized_token_match'] for r in results),'differences':[r for r in results if not r['normalized_token_match']],'files_without_prior_RE_mapping':sorted(all_current-{x[0] for x in pairs}),'all_pairs':results}
(out/'RE_SOURCE_INVENTORY.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'mapped_files':report['mapped_files'],'normalized_matches':report['normalized_matches'],'unmapped_count':len(report['files_without_prior_RE_mapping']),'unmapped':report['files_without_prior_RE_mapping']},indent=2))
print('Changed mapped files:')
for row in report['differences']:print(row['current'],'<-',row['original'])

def area(name,role):
 stem=Path(name).stem
 if '/transport_' in name or (('/xml/' in name) and stem not in ('reader','memory_printer')):return 'REFERENCE_TRANSPORT_XML_MODELS'
 if stem in ('bootstrap','recovered_effects','thread_launch','hook_cleanup','hook_engine_adapters','pipeline'):return 'REFERENCE_COMPOSITION_ADAPTERS'
 if stem in ('application','application_internal','application_defaults','dll_entry','control') and '/config/' not in name:return 'APPLICATION_STARTUP_SHUTDOWN'
 if stem=='application_environment':return 'ENVIRONMENT_OS_BINDINGS'
 if '/config/' in name:return 'CONFIGURATION_VDF_HASH_ORDER'
 if stem in ('atomic_files','atomic_files_windows'):return 'ATOMIC_PERSISTENCE_OWNERSHIP'
 if stem in ('file_copy','file_copy_windows'):return 'RECOVERED_FILE_COPY'
 if stem in ('execution','execution_windows','startup','startup_windows') and '/runtime/' in name or stem in ('execution_windows','startup_windows'):return 'EXECUTION_ARGUMENTS_FLAGS'
 if stem.startswith(('hook_transaction','hook_unload_guard','hook_engine_api')):return 'HOOK_INSTALL_ROLLBACK_UNLOAD'
 if stem in ('hook_native_callbacks','hook_original_effects','hook_setup','callback_lock'):return 'CALLBACK_BINDING_LEASE_LOCKS'
 if stem in ('hook_connect',):return 'CONNECT_ADDRESS_REDIRECTION'
 if stem in ('crypt_query','crypt_query_host'):return 'HOOK11_UTF16_FORWARDING'
 if stem in ('registry','hook_host_callbacks'):return 'REGISTRY_PROCESS_FOLDER'
 if '/net/' in name or '/protocol/' in name:return 'LIVE_TRANSPORT_MESSAGES'
 if stem in ('event_queue','event_adapters'):return 'EVENT_QUEUE_RESERVATIONS_INITIALIZATION'
 if stem.startswith('achievement'):return 'ACHIEVEMENT_STATE_STORAGE'
 if stem in ('calendar','timestamp'):return 'TIME_CONVERSION'
 if '/handlers/' in name or '/router/' in name:return 'HANDLERS_ROUTER_SERIALIZATION'
 if stem in ('online','libraries','library_system'):return 'PROVIDER_EXTRA_LIBRARY_BINDINGS'
 if stem in ('product_identity','protocol_identity'):return 'FIXED_PRODUCT_PROTOCOL_IDENTITIES'
 if '/codec/' in name or '/hash/' in name:return 'CODEC_HASH_ENCODING'
 if '/xml/' in name:return 'VENDORED_XML_BRIDGE_PRINTER'
 return 'FILE_PATH_STATUS_LICENSE_TOKEN'
