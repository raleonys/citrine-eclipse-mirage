cmake_minimum_required(VERSION 3.20)
project(xml_backend_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(XML_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT XML_REFERENCE_ROOT)
  message(FATAL_ERROR "Provide XML_REFERENCE_ROOT (pinned PR #108 control)")
endif()
if(NOT PUGIXML_ROOT)
  set(PUGIXML_ROOT "${XML_PROJECT_ROOT}/third_party/pugixml")
endif()
file(TO_CMAKE_PATH "${XML_REFERENCE_ROOT}" XML_REFERENCE_ROOT)
file(TO_CMAKE_PATH "${PUGIXML_ROOT}" PUGIXML_ROOT)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(XML_COMPARISON_SANITIZERS "Use ASan and UBSan in this standalone diagnostic" OFF)
if(XML_COMPARISON_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "The ASan/UBSan diagnostic profile requires GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/run.py"
  --reference-root "${XML_REFERENCE_ROOT}" --pugixml-root "${PUGIXML_ROOT}" --verify-only
  RESULT_VARIABLE input_status)
if(NOT input_status EQUAL 0)
  message(FATAL_ERROR "XML comparison input identity check failed")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/reference_bindings.cmake")

# The retired vendor exists only in the external, immutable reference checkout.
add_library(comparison_tiny STATIC "${XML_REFERENCE_ROOT}/third_party/tinyxml2/tinyxml2.cpp")
target_include_directories(comparison_tiny PUBLIC "${XML_REFERENCE_ROOT}/third_party/tinyxml2")
add_library(comparison_pugi STATIC "${PUGIXML_ROOT}/src/pugixml.cpp")
target_include_directories(comparison_pugi PUBLIC "${PUGIXML_ROOT}/src")
target_compile_definitions(comparison_pugi PUBLIC PUGIXML_NO_XPATH)
add_library(comparison_common STATIC
  "${XML_PROJECT_ROOT}/src/xml/memory_printer.cpp"
  "${XML_PROJECT_ROOT}/src/protocol/messages.cpp")
target_include_directories(comparison_common PUBLIC "${XML_PROJECT_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
# Only rename the first-party oracle namespace, avoiding an ODR collision with
# the production Backend. Its assertions and all original vendor bytes remain.
xml_read_reference_header("${XML_REFERENCE_ROOT}" oracle_header)
string(REPLACE "laureondo::xml_reader_bridge" "laureondo::xml_tiny_reference" oracle_header "${oracle_header}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/oracle_reader.hpp" "${oracle_header}")
file(READ "${XML_REFERENCE_ROOT}/src/xml/reader.cpp" oracle_source)
xml_bind_legacy_reference(oracle_source)
string(REPLACE "laureondo/xml/reader.hpp" "oracle_reader.hpp" oracle_source "${oracle_source}")
string(REPLACE "laureondo::xml_reader_bridge" "laureondo::xml_tiny_reference" oracle_source "${oracle_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/oracle_reader.cpp" "${oracle_source}")
add_library(comparison_reference STATIC "${CMAKE_CURRENT_BINARY_DIR}/oracle_reader.cpp")
target_include_directories(comparison_reference PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(comparison_reference PUBLIC comparison_common comparison_tiny)
# Exactly the selected first-party production reader sources, never test copies.
add_library(comparison_candidate STATIC
  "${XML_PROJECT_ROOT}/src/xml/reader.cpp"
  "${XML_PROJECT_ROOT}/src/xml/reader_policy.cpp"
  "${XML_PROJECT_ROOT}/src/xml/reader_values.cpp")
target_link_libraries(comparison_candidate PUBLIC comparison_common comparison_pugi)
add_library(comparison_storage STATIC
  "${XML_PROJECT_ROOT}/src/runtime/achievement_storage.cpp"
  "${XML_PROJECT_ROOT}/src/runtime/achievement_writer.cpp"
  "${XML_PROJECT_ROOT}/src/core/binary_io.cpp"
  "${XML_PROJECT_ROOT}/src/core/paths.cpp")
target_include_directories(comparison_storage PUBLIC "${XML_PROJECT_ROOT}/include")

file(READ "${XML_REFERENCE_ROOT}/tests/unit/messages/messages_tests.cpp" messages_fixture)
xml_bind_legacy_reference(messages_fixture)
string(FIND "${messages_fixture}" "\nint main()" entrypoint)
if(entrypoint LESS 0)
  message(FATAL_ERROR "Messages fixture extraction marker changed")
endif()
string(SUBSTRING "${messages_fixture}" 0 ${entrypoint} messages_fixture)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/messages_fixtures.hpp" "${messages_fixture}\n")
file(READ "${XML_REFERENCE_ROOT}/tests/unit/achievement_storage/storage_fixture.hpp" storage_fixture)
xml_bind_legacy_reference(storage_fixture)
string(REPLACE "namespace {" "namespace eval_storage {" storage_fixture "${storage_fixture}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/storage_fixtures.hpp" "${storage_fixture}")
add_executable(comparison_suite suite.cpp)
if(MSVC)
  target_compile_options(comparison_suite PRIVATE /we4715)
else()
  target_compile_options(comparison_suite PRIVATE -Werror=return-type)
endif()
target_include_directories(comparison_suite PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(comparison_suite PRIVATE comparison_candidate comparison_reference comparison_storage)
add_executable(comparison_allocations allocation.cpp)
target_link_libraries(comparison_allocations PRIVATE comparison_candidate comparison_reference)
foreach(test IN ITEMS reader corpus)
  file(READ "${XML_REFERENCE_ROOT}/tests/unit/xml_reader/${test}_tests.cpp" reference_tests)
  xml_bind_legacy_reference(reference_tests)
  string(REPLACE "laureondo/xml/reader.hpp" "oracle_reader.hpp" reference_tests "${reference_tests}")
  string(REPLACE "laureondo::xml_reader_bridge" "laureondo::xml_tiny_reference" reference_tests "${reference_tests}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference_${test}.cpp" "${reference_tests}")
  add_executable(reference_${test} "${CMAKE_CURRENT_BINARY_DIR}/reference_${test}.cpp")
  target_link_libraries(reference_${test} PRIVATE comparison_reference)
  add_executable(candidate_${test} "${XML_PROJECT_ROOT}/tests/unit/xml_reader/${test}_tests.cpp")
  target_link_libraries(candidate_${test} PRIVATE comparison_candidate)
endforeach()
add_executable(candidate_allocations "${XML_PROJECT_ROOT}/tests/performance/xml_reader_probe.cpp")
target_link_libraries(candidate_allocations PRIVATE comparison_candidate)

if(WIN32)
  add_library(comparison_windows_services STATIC
    "${XML_PROJECT_ROOT}/src/codec/crypto.cpp"
    "${XML_PROJECT_ROOT}/src/router/router.cpp"
    "${XML_PROJECT_ROOT}/src/handlers/handlers.cpp"
    "${XML_PROJECT_ROOT}/src/handlers/misc_handlers.cpp"
    "${XML_PROJECT_ROOT}/src/handlers/content_handlers.cpp"
    "${XML_PROJECT_ROOT}/src/core/license.cpp"
    "${XML_PROJECT_ROOT}/src/core/atomic_files.cpp"
    "${XML_PROJECT_ROOT}/src/platform/file_io_windows.cpp"
    "${XML_PROJECT_ROOT}/src/platform/atomic_files_windows.cpp")
  target_link_libraries(comparison_windows_services PUBLIC comparison_common comparison_storage)
  foreach(test IN ITEMS dispatch storage)
    if(test STREQUAL "dispatch")
      set(test_path "tests/unit/dispatch/integration_tests.cpp")
    else()
      set(test_path "tests/unit/achievement_storage/storage_tests.cpp")
    endif()
    file(READ "${XML_REFERENCE_ROOT}/${test_path}" reference_tests)
    xml_bind_legacy_reference(reference_tests)
    string(REPLACE "laureondo/xml/reader.hpp" "oracle_reader.hpp" reference_tests "${reference_tests}")
    string(REPLACE "laureondo::xml_reader_bridge" "laureondo::xml_tiny_reference" reference_tests "${reference_tests}")
    string(REPLACE "xml_reader_bridge::Backend" "xml_tiny_reference::Backend" reference_tests "${reference_tests}")
    string(REPLACE "\"storage_fixture.hpp\"" "\"reference_storage_fixture.hpp\"" reference_tests "${reference_tests}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference_${test}.cpp" "#include \"oracle_reader.hpp\"\n${reference_tests}")
    add_executable(reference_${test} "${CMAKE_CURRENT_BINARY_DIR}/reference_${test}.cpp")
    target_link_libraries(reference_${test} PRIVATE comparison_reference comparison_windows_services)
    add_executable(candidate_${test} "${XML_PROJECT_ROOT}/${test_path}")
    target_link_libraries(candidate_${test} PRIVATE comparison_candidate comparison_windows_services)
  endforeach()
  file(READ "${XML_REFERENCE_ROOT}/tests/unit/achievement_storage/storage_fixture.hpp" reference_storage_fixture)
  xml_bind_legacy_reference(reference_storage_fixture)
  string(REPLACE "laureondo/xml/reader.hpp" "oracle_reader.hpp" reference_storage_fixture "${reference_storage_fixture}")
  string(REPLACE "xml_reader_bridge::Backend" "xml_tiny_reference::Backend" reference_storage_fixture "${reference_storage_fixture}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference_storage_fixture.hpp" "${reference_storage_fixture}")
endif()
